cmake_minimum_required(VERSION 3.24)
project(D5Render LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_DEBUG ${CMAKE_BINARY_DIR}/bin)
set(TRT_DIR "C:/TensorRT")
set(CUDA_DIR "C:/Program Files/NVIDIA GPU Computing Toolkit/CUDA/v13.0")
set(ORT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/onnxruntime_gpu/onnxruntime-win-x64-gpu-1.26.0")
set(ORT_DML_DIR "${CMAKE_CURRENT_SOURCE_DIR}/onnxruntime_dml")
set(DML_DIR "${CMAKE_CURRENT_SOURCE_DIR}/directml_sdk")
set(CUDNN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cudnn-windows-x86_64-9.24.0.43_cuda13-archive")
include(FetchContent)
set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build/_deps/imgui-src")
set(CUDA_OBJ ${CMAKE_CURRENT_BINARY_DIR}/cuda_preprocess.obj)
if(EXISTS "${CUDA_OBJ}")
    message(STATUS "使用预编译的CUDA对象文件: ${CUDA_OBJ}")
else()
    add_custom_command(
        OUTPUT ${CUDA_OBJ}
        COMMAND "${CUDA_DIR}/bin/nvcc.exe" -allow-unsupported-compiler -c "${CMAKE_CURRENT_SOURCE_DIR}/inference/cuda_preprocess.cu" -o "${CUDA_OBJ}" -I"${CUDA_DIR}/include" -I"${CMAKE_CURRENT_SOURCE_DIR}" --std=c++17 -Xcompiler "/EHsc /MT" -DNOMINMAX -DUNICODE -D_UNICODE
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/inference/cuda_preprocess.cu" "${CMAKE_CURRENT_SOURCE_DIR}/inference/cuda_preprocess.h"
        COMMENT "编译CUDA预处理内核"
    )
endif()
set(SOURCES main.cpp config/config.cpp capture/screen_capture.cpp inference/trt_engine.cpp inference/dml_engine.cpp inference/yolo_postprocess.cpp inference/onnx_metadata.cpp inference/gpu_perf.cpp aim/aim_assist.cpp input/logitech_driver.cpp input/makcu_driver.cpp gui/app.cpp utils/update_checker.cpp utils/gpu_clock.cpp ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp ${IMGUI_DIR}/backends/imgui_impl_win32.cpp)
add_executable(${PROJECT_NAME} WIN32 ${SOURCES} resource/dd.rc ${CUDA_OBJ})
target_include_directories(${PROJECT_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CUDA_DIR}/include ${TRT_DIR}/include ${IMGUI_DIR} ${IMGUI_DIR}/backends ${ORT_DIR}/include ${ORT_DML_DIR}/build/native/include ${DML_DIR}/include)
target_link_libraries(${PROJECT_NAME} PRIVATE ${CUDA_DIR}/lib/x64/cudart.lib ${CUDA_DIR}/lib/x64/cuda.lib ${CUDA_DIR}/lib/x64/cublas.lib ${CUDA_DIR}/lib/x64/cublasLt.lib ${TRT_DIR}/lib/nvinfer_10.lib ${TRT_DIR}/lib/nvonnxparser_10.lib ${TRT_DIR}/lib/nvinfer_plugin_10.lib ${ORT_DIR}/lib/onnxruntime.lib ${ORT_DIR}/lib/onnxruntime_providers_cuda.lib ${ORT_DIR}/lib/onnxruntime_providers_tensorrt.lib ${ORT_DIR}/lib/onnxruntime_providers_shared.lib ${DML_DIR}/bin/x64-win/DirectML.lib d3d11 dxgi d3dcompiler dwmapi user32 gdi32 comctl32 shell32 runtimeobject wintrust crypt32 setupapi iphlpapi winhttp winmm)
target_compile_definitions(${PROJECT_NAME} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINDOWS_FOUNDATION_UNIVERSALAPICONTRACT_VERSION=0x130000)
target_compile_options(${PROJECT_NAME} PRIVATE /MP /EHsc /MT /utf-8 /Zi)
target_link_options(${PROJECT_NAME} PRIVATE /DEBUG /PDB:${CMAKE_BINARY_DIR}/bin/D5Render.pdb)
file(GLOB TRT_DLLS "${TRT_DIR}/bin/*.dll")
file(GLOB CUDA_DLLS "${CUDA_DIR}/bin/x64/cudart*.dll" "${CUDA_DIR}/bin/x64/cublas*.dll" "${CUDA_DIR}/bin/x64/cublasLt*.dll" "${CUDA_DIR}/bin/x64/cufft*.dll" "${CUDA_DIR}/bin/x64/curand*.dll")
set(CUDA_DRIVER_DLL "")
set(ORT_DLL "${ORT_DIR}/runtimes/win-x64/native/onnxruntime.dll")
set(DML_DLL "${DML_DIR}/bin/x64-win/DirectML.dll")
file(GLOB ORT_GPU_DLLS "${ORT_DIR}/lib/*.dll")
file(GLOB CUDNN_DLLS "${CUDNN_DIR}/bin/x64/*.dll")
set(GHUB_SEARCH_PATHS
    "${CMAKE_CURRENT_SOURCE_DIR}/driver"
    "$ENV{USERPROFILE}/Downloads/logitech.driver"
    "C:/Program Files/LGHUB"
    "C:/Program Files (x86)/LGHUB"
    "C:/Program Files/Logitech/Logitech G HUB"
    "C:/Program Files (x86)/Logitech/Logitech G HUB"
)
set(GHUB_DLL_NAMES logitech.driver.dll ghub_device.dll LogitechGHUBDevice.dll LGHUBDevice.dll)
set(GHUB_DLL "GHUB_DLL-NOTFOUND")
foreach(search_path ${GHUB_SEARCH_PATHS})
    foreach(dll_name ${GHUB_DLL_NAMES})
        if(EXISTS "${search_path}/${dll_name}")
            set(GHUB_DLL "${search_path}/${dll_name}")
            message(STATUS "找到GHUB驱动DLL: ${GHUB_DLL}")
            break()
        endif()
    endforeach()
    if(NOT GHUB_DLL STREQUAL "GHUB_DLL-NOTFOUND")
        break()
    endif()
endforeach()
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${TRT_DLLS} ${CUDA_DLLS} ${ORT_GPU_DLLS} ${CUDNN_DLLS} ${DML_DLL} "$<TARGET_FILE_DIR:${PROJECT_NAME}>")
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/doc/usage.html" "$<TARGET_FILE_DIR:${PROJECT_NAME}>/使用说明书.html" COMMENT "复制使用说明书")
if(EXISTS "${CUDA_DRIVER_DLL}" AND NOT CUDA_DRIVER_DLL STREQUAL "")
    add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CUDA_DRIVER_DLL}" "$<TARGET_FILE_DIR:${PROJECT_NAME}>" COMMENT "复制CUDA Driver DLL")
endif()
if(NOT GHUB_DLL STREQUAL "GHUB_DLL-NOTFOUND")
    add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GHUB_DLL}" "$<TARGET_FILE_DIR:${PROJECT_NAME}>" COMMENT "复制GHUB驱动DLL: ${GHUB_DLL}")
    message(STATUS "构建后将自动复制GHUB驱动DLL到输出目录")
else()
    message(WARNING "未找到GHUB驱动DLL，罗技驱动功能将不可用(可使用通用SendInput驱动)")
endif()
